FROM python:3.11-slim

WORKDIR /app

# System dependencies required by opencv and image processing
RUN apt-get update && apt-get install -y --no-install-recommends \
    libgl1 \
    libglib2.0-0 \
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

# Default fallback port if PORT is not set by the host environment
ENV PORT=10000
EXPOSE 10000

# Shell form CMD ensures $PORT is expanded dynamically at container launch
CMD uvicorn app.main:app --host 0.0.0.0 --port $PORT
